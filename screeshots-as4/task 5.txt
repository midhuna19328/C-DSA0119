#include <iostream>
#include <iomanip>
#include <string>
using namespace std;
class Cart {
string items[10];
double prices[10];
int n = 0;
double discount = 0;
public:
// TODO 1: Add item and return *this
Cart& add(string item, double price) {
if (n < 10) {
items[n] = item;

prices[n] = price;
n++;
}
return *this;
}

// TODO 2: Set discount using this pointer
Cart& setDiscount(double discount) {
this->discount = discount;
return *this;
}
double total() const {
double t = 0;
for (int i = 0; i < n; ++i)
t += prices[i];
return t * (100 - discount) / 100;
}
};
int main() {
// (a) Direct access using object name
Cart c;
c.add("Rice", 60).add("Dal", 120).setDiscount(10);
// (b) Access using pointer
Cart* p = &c;
p->add("Oil", 180);
cout << fixed << setprecision(2);
cout << "Total using pointer: " << p->total() << endl;

// (c) Access using reference
Cart& r = c;
cout << "Total using reference: " << r.total() << endl;
// (d) Dynamic object using new and delete
Cart* d = new Cart;
d->add("Milk", 50).add("Bread", 40);
cout << "Dynamic cart total: " << d->total() << endl;
delete d;
return 0;
}
